    41→function emptyResources(): Resources {
    42→  return { food: 0, water: 0, materials: 0, medicine: 0, fuel: 0, ammo: 0 };

    45→function clampResources(res: Resources, caps: Resources): Resources {
    46→  const out: Resources = { ...res };
    47→  (Object.keys(out) as ResourceType[]).forEach((k) => {
    48→    out[k] = Math.max(0, Math.min(caps[k], out[k]));
    49→  });
    50→  return out;
    51→}
    52→
    53→function canAfford(res: Resources, cost: Partial<Resources>): boolean {
    54→  for (const [k, v] of Object.entries(cost)) {
    55→    if (res[k as ResourceType] < (v as number)) return false;
    56→  }
    57→  return true;
    58→}
    59→
    60→function subtractCost(res: Resources, cost: Partial<Resources>): Resources {
    61→  const out = { ...res };
    62→  for (const [k, v] of Object.entries(cost)) {
    63→    out[k as ResourceType] -= v as number;
    64→  }
    65→  return out;
    66→}

   108→function calculateCaps(area: Area): Resources {
   109→  if (!area.hasBase) return { ...BASELESS_CAPS };
   110→  const base = { ...INITIAL_RESOURCE_CAPS };
   111→  const storageLevel = area.buildings.storage.level;
   112→  (Object.keys(base) as ResourceType[]).forEach((k) => {
   113→    base[k] += storageLevel * 50;
   114→  });
   115→  return base;
   116→}

   179→// ---------- Resolve a scout Mission (combat + loot) ----------
   180→function resolveMission(
   181→  mission: Mission,
   182→  survivors: Survivor[],
   183→  location: GameLocation,
   184→  day: number
   185→): { result: MissionResult; log: GameLogEntry[] } {
   186→  const log: GameLogEntry[] = [];
   187→  const teamSurvivors = survivors.filter((s) => mission.team.includes(s.id));
   188→  const rng = makeRng(
   189→    mission.id
   190→      .split("")
   191→      .reduce((a, c) => a + c.charCodeAt(0), 0) +
   192→      day * 7919 +
   193→      location.position[0] * 31
   194→  );
   195→
   196→  const result: MissionResult = {
   197→    success: true,
   198→    lootGained: {},
   199→    survivorsRecruited: [],
   200→    casualties: [],
   201→    injuries: [],
   202→    log: [],
   203→  };
   204→
   205→  if (!location.cleared && location.enemyCount > 0) {
   206→    const enemyDef = ENEMY_INFO[location.enemyType];
   207→    const teamCombat =
   208→      teamSurvivors.reduce((sum, s) => sum + s.skills.combat, 0) +
   209→      teamSurvivors.reduce((sum, s) => sum + (s.health > 50 ? 1 : 0), 0) * 0.5;
   210→    const enemyPower = location.enemyCount * enemyDef.combatPower;
   211→
   212→    result.log.push(
   213→      `Team encountered ${location.enemyCount} ${enemyDef.label} at ${location.name}.`
   214→    );
   215→
   216→    if (teamCombat >= enemyPower) {
   217→      result.success = true;
   218→      result.log.push(
   219→        `Victory! Team combat ${teamCombat.toFixed(1)} vs enemy ${enemyPower.toFixed(1)}.`
   220→      );
   221→      teamSurvivors.forEach((s) => {
   222→        const injuryChance = 0.15 + location.dangerLevel * 0.05;
   223→        if (rng() < injuryChance) {
   224→          const dmg = randInt(rng, 10, 30);
   225→          s.health = Math.max(1, s.health - dmg);
   226→          s.status = s.health < 30 ? "critical" : "injured";
   227→          result.injuries.push(s.id);
   228→          result.log.push(`${s.name} was injured (-${dmg} HP).`);
   229→        }
   230→      });
   231→    } else {
   232→      result.success = false;
   233→      result.log.push(
   234→        `Defeat! Team combat ${teamCombat.toFixed(1)} vs enemy ${enemyPower.toFixed(1)}.`
   235→      );
   236→      teamSurvivors.forEach((s) => {
   237→        const deathChance = 0.1 + Math.max(0, enemyPower - teamCombat) * 0.05;
   238→        if (rng() < deathChance) {
   239→          result.casualties.push(s.id);
   240→          result.log.push(`${s.name} was killed in action.`);
   241→        } else {
   242→          const dmg = randInt(rng, 25, 55);
   243→          s.health = Math.max(5, s.health - dmg);
   244→          s.status = s.health < 30 ? "critical" : "injured";
   245→          result.injuries.push(s.id);
   246→          result.log.push(`${s.name} was badly injured (-${dmg} HP).`);
   247→        }
   248→      });
   249→      const halfLoot: Partial<Resources> = {};
   250→      for (const [k, v] of Object.entries(location.loot)) {
   251→        halfLoot[k as ResourceType] = Math.floor((v as number) * 0.4);
   252→      }
   253→      result.lootGained = halfLoot;
   254→    }
   255→  } else {
   256→    result.log.push(`No hostiles at ${location.name}.`);
   257→  }
   258→
   259→  if (result.success && !location.cleared) {
   260→    const scavenging = teamSurvivors.reduce(
   261→      (sum, s) => sum + s.skills.scavenging,
   262→      0
   263→    );
   264→    const lootMultiplier = 0.5 + scavenging * 0.15 + rng() * 0.3;
   265→    const loot: Partial<Resources> = {};
   266→    for (const [k, v] of Object.entries(location.loot)) {
   267→      loot[k as ResourceType] = Math.max(
   268→        0,
   269→        Math.round((v as number) * lootMultiplier)
   270→      );
   271→    }
   272→    result.lootGained = { ...result.lootGained, ...loot };
   273→    const lootSummary = Object.entries(loot)
   274→      .filter(([, v]) => (v as number) > 0)
   275→      .map(([k, v]) => `${v} ${k}`)
   276→      .join(", ");
   277→    if (lootSummary) {
   278→      result.log.push(`Salvaged: ${lootSummary}.`);
   279→    }
   280→  }
   281→
   282→  if (result.success && !location.cleared) {
   283→    if (rng() < location.survivorChance) {
   284→      const newSurvivor = generateSurvivor(
   285→        Math.floor(rng() * 1000000) + day * 13
   286→      );
   287→      newSurvivor.health = randInt(rng, 50, 90);
   288→      newSurvivor.hunger = randInt(rng, 30, 70);
   289→      newSurvivor.thirst = randInt(rng, 30, 70);
   290→      newSurvivor.morale = randInt(rng, 30, 70);
   291→      result.survivorsRecruited.push(newSurvivor);
   292→      result.log.push(
   293→        `Found a survivor named ${newSurvivor.name}! They joined your group.`
   294→      );
   295→    }
   296→  }
   297→
   298→  location.explored = true;
   299→  if (result.success) {
   300→    location.cleared = true;
   301→    location.enemyCount = 0;
   302→  }
   303→
   304→  result.log.forEach((msg) =>
   305→    log.push({
   306→      day,
   307→      message: msg,
   308→      type: msg.includes("killed")
   309→        ? "danger"
   310→        : msg.includes("injured") || msg.includes("Defeat")
   311→        ? "warning"
   312→        : msg.includes("Victory") ||
   313→          msg.includes("Salvaged") ||
   314→          msg.includes("Found")
   315→        ? "success"
   316→        : "info",
   317→    })
   318→  );
   319→
   320→  return { result, log };
   321→}

   323→// ---------- Process a single area for the day ----------
   324→function processArea(
   325→  area: Area,
   326→  allSurvivors: Record<string, Survivor>,
   327→  day: number,
   328→  newLog: GameLogEntry[]
   329→): { survivorsDied: string[] } {
   330→  const survivorsDied: string[] = [];
   331→
   332→  if (!area.hasBase && area.survivorIds.length === 0 && area.missions.length === 0) {
   333→    return { survivorsDied };
   334→  }
   335→
   336→  const areaSurvivors = area.survivorIds
   337→    .map((id) => allSurvivors[id])
   338→    .filter((s) => s !== undefined);
   339→  let resources = { ...area.resources };
   340→
   341→  newLog.push({
   342→    day,
   343→    message: `--- ${area.name} (Day ${day}) ---`,
   344→    type: "info",
   345→  });
   346→
   347→  // ---------- 1. Resolve pending missions ----------
   348→  area.missions = area.missions.map((m) => {
   349→    if (m.status !== "pending") return m;
   350→    const location = area.locations.find((l) => l.id === m.locationId);
   351→    if (!location) return { ...m, status: "completed" as const };
   352→
   353→    const teamSurvivors = m.team
   354→      .map((id) => allSurvivors[id])
   355→      .filter((s) => s !== undefined);
   356→
   357→    if (m.missionType !== "salvage") {
   358→      teamSurvivors.forEach((s) => {
   359→        s.role = "onMission";
   360→      });
   361→    }
   362→
   363→    // -------- Salvage mission --------
   364→    if (m.missionType === "salvage") {
   365→      const salvageRng = makeRng(
   366→        m.id.split("").reduce((a, c) => a + c.charCodeAt(0), 0) + day * 7919
   367→      );
   368→      const totalScavenge = teamSurvivors.reduce(
   369→        (sum, s) => sum + s.skills.scavenging,
   370→        0
   371→      );
   372→      const salvageWork = totalScavenge + teamSurvivors.length * 2;
   373→      const poolEntries = Object.entries(location.salvagePool).filter(
   374→        ([, v]) => (v as number) > 0
   375→      ) as [ResourceType, number][];
   376→
   377→      const lootGained: Partial<Resources> = {};
   378→      let depletedThisRun = false;
   379→
   380→      if (poolEntries.length === 0) {
   381→        location.salvageDepleted = true;
   382→        depletedThisRun = true;
   383→      } else {
   384→        const hasMaterials = poolEntries.some(([k]) => k === "materials");
   385→        const otherEntries = poolEntries.filter(([k]) => k !== "materials");
   386→        const allocations: [ResourceType, number][] = [];
   387→        if (hasMaterials) {
   388→          allocations.push(["materials", salvageWork * 0.5]);
   389→        }
   390→        if (otherEntries.length > 0) {
   391→          const perType = (salvageWork * (hasMaterials ? 0.5 : 1)) / otherEntries.length;
   392→          for (const [k] of otherEntries) {
   393→            allocations.push([k, perType]);
   394→          }
   395→        }
   396→        for (const [k, work] of allocations) {
   397→          const remaining = location.salvagePool[k] ?? 0;
   398→          if (remaining <= 0) continue;
   399→          const yield_ = Math.max(1, Math.round(work * (0.8 + salvageRng() * 0.4)));
   400→          const extracted = Math.min(remaining, yield_);
   401→          location.salvagePool[k] = remaining - extracted;
   402→          lootGained[k] = extracted;
   403→        }
   404→        const totalRemaining = Object.values(location.salvagePool).reduce(
   405→          (sum, v) => sum + (v as number),
   406→          0
   407→        );
   408→        if (totalRemaining <= 0) {
   409→          location.salvageDepleted = true;
   410→          depletedThisRun = true;
   411→        }
   412→        const lootSummary = Object.entries(lootGained)
   413→          .filter(([, v]) => (v as number) > 0)
   414→          .map(([k, v]) => `${v} ${k}`)
   415→          .join(", ");
   416→        if (lootSummary) {
   417→          newLog.push({
   418→            day,
   419→            message: `[${area.name}] Salvaged: ${lootSummary}.`,
   420→            type: "success",
   421→          });
   422→        }
   423→        if (depletedThisRun) {
   424→          newLog.push({
   425→            day,
   426→            message: `[${area.name}] ${location.name} fully salvaged.`,
   427→            type: "warning",
   428→          });
   429→        }
   430→      }
   431→
   432→      for (const [k, v] of Object.entries(lootGained)) {
   433→        resources[k as ResourceType] += v as number;
   434→      }
   435→
   436→      if (depletedThisRun) {
   437→        teamSurvivors.forEach((s) => {
   438→          s.role = "idle";
   439→          s.assignedTeamId = undefined;
   440→        });
   441→        const team = area.teams.find((t) => t.id === m.teamId);
   442→        if (team) team.locationId = null;
   443→        return { ...m, status: "completed" as const };
   444→      }
   445→      return m; // stays pending
   446→    }
   447→
   448→    // -------- Scout mission --------
   449→    const { result, log } = resolveMission(m, teamSurvivors, location, day);
   450→    newLog.push(...log);
   451→
   452→    for (const [k, v] of Object.entries(result.lootGained)) {
   453→      resources[k as ResourceType] += v as number;
   454→    }
   455→
   456→    result.casualties.forEach((id) => survivorsDied.push(id));
   457→    result.survivorsRecruited.forEach((ns) => {
   458→      allSurvivors[ns.id] = ns;
   459→      area.survivorIds.push(ns.id);
   460→    });
   461→
   462→    teamSurvivors.forEach((s) => {
   463→      const updated = allSurvivors[s.id];
   464→      if (updated) {
   465→        updated.role = "idle";
   466→        updated.assignedTeamId = undefined;
   467→      }
   468→    });
   469→
   470→    const team = area.teams.find((t) => t.id === m.teamId);
   471→    if (team) team.locationId = null;
   472→
   473→    return { ...m, status: "completed" as const, result };
   474→  });
   475→
   476→  area.missions = area.missions.filter((m) => m.status === "pending");
   477→
   478→  // ---------- 2. Building production ----------
   479→  if (area.hasBase) {
   480→    const farmFood = area.buildings.farm.level * 5;
   481→    const wellWater = area.buildings.well.level * 5;
   482→    if (farmFood > 0) {
   483→      resources.food += farmFood;
   484→      newLog.push({ day, message: `[${area.name}] Farm produced ${farmFood} food.`, type: "success" });
   485→    }
   486→    if (wellWater > 0) {
   487→      resources.water += wellWater;
   488→      newLog.push({ day, message: `[${area.name}] Well produced ${wellWater} water.`, type: "success" });
   489→    }
   490→  }
   491→
   492→  // ---------- 3. Survivor consumption ----------
   493→  const consumption = areaSurvivors.length;
   494→  const foodNeeded = consumption;
   495→  const waterNeeded = consumption;
   496→  let foodShortage = 0;
   497→  let waterShortage = 0;
   498→
   499→  if (resources.food >= foodNeeded) {
   500→    resources.food -= foodNeeded;
   501→  } else {
   502→    foodShortage = foodNeeded - resources.food;
   503→    resources.food = 0;
   504→  }
   505→  if (resources.water >= waterNeeded) {
   506→    resources.water -= waterNeeded;
   507→  } else {
   508→    waterShortage = waterNeeded - resources.water;
   509→    resources.water = 0;
   510→  }
   511→
   512→  if (foodShortage > 0) {
   513→    newLog.push({ day, message: `[${area.name}] Food shortage! ${foodShortage} survivors go hungry.`, type: "warning" });
   514→  } else if (consumption > 0) {
   515→    newLog.push({ day, message: `[${area.name}] Survivors consumed ${foodNeeded} food and ${waterNeeded} water.`, type: "info" });
   516→  }
   517→  if (waterShortage > 0) {
   518→    newLog.push({ day, message: `[${area.name}] Water shortage! ${waterShortage} survivors are dehydrated.`, type: "warning" });
   519→  }
   520→
   521→  // ---------- 4. Update survivor stats ----------
   522→  areaSurvivors.forEach((s) => {
   523→    if (foodShortage === 0 && s.role !== "onMission") {
   524→      s.hunger = Math.max(0, s.hunger - 30);
   525→    } else if (s.role === "onMission") {
   526→      s.hunger = Math.min(100, s.hunger + 10);
   527→    } else {
   528→      s.hunger = Math.min(100, s.hunger + 25);
   529→      s.health = Math.max(0, s.health - 5);
   530→    }
   531→
   532→    if (waterShortage === 0 && s.role !== "onMission") {
   533→      s.thirst = Math.max(0, s.thirst - 35);
   534→    } else if (s.role === "onMission") {
   535→      s.thirst = Math.min(100, s.thirst + 15);
   536→    } else {
   537→      s.thirst = Math.min(100, s.thirst + 30);
   538→      s.health = Math.max(0, s.health - 8);
   539→    }
   540→
   541→    if (s.hunger >= 90) {
   542→      s.health = Math.max(0, s.health - 8);
   543→      newLog.push({ day, message: `[${area.name}] ${s.name} is starving.`, type: "warning" });
   544→    }
   545→    if (s.thirst >= 90) {
   546→      s.health = Math.max(0, s.health - 10);
   547→      newLog.push({ day, message: `[${area.name}] ${s.name} is severely dehydrated.`, type: "warning" });
   548→    }
   549→
   550→    if (s.role === "resting") {
   551→      s.morale = Math.min(100, s.morale + 10);
   552→      s.health = Math.min(100, s.health + 5);
   553→    }
   554→    if (s.health < 30) {
   555→      s.morale = Math.max(0, s.morale - 5);
   556→    }
   557→  });
   558→
   559→  // ---------- 5. Healing ----------
   560→  if (area.hasBase && area.buildings.infirmary.level > 0) {
   561→    const heal = area.buildings.infirmary.level * 10;
   562→    areaSurvivors.forEach((s) => {
   563→      if (s.status !== "healthy" && s.health < 100) {
   564→        if (resources.medicine >= 1) {
   565→          resources.medicine -= 1;
   566→          s.health = Math.min(100, s.health + heal);
   567→          if (s.health >= 70) s.status = "healthy";
   568→          else if (s.health >= 40 && s.status === "critical") s.status = "injured";
   569→          newLog.push({ day, message: `[${area.name}] ${s.name} treated at infirmary (+${heal} HP).`, type: "success" });
   570→        }
   571→      }
   572→    });
   573→  } else {
   574→    areaSurvivors.forEach((s) => {
   575→      if (s.status !== "healthy" && s.health < 100) {
   576→        const naturalHeal = 3;
   577→        s.health = Math.min(100, s.health + naturalHeal);
   578→        if (s.health >= 70) s.status = "healthy";
   579→        else if (s.health >= 40 && s.status === "critical") s.status = "injured";
   580→      }
   581→    });
   582→  }
   583→
   584→  // Update status
   585→  areaSurvivors.forEach((s) => {
   586→    if (s.health <= 0) {
   587→      // dead
   588→    } else if (s.health < 25) {
   589→      s.status = "critical";
   590→    } else if (s.health < 50) {
   591→      s.status = s.status === "sick" ? "sick" : "injured";
   592→    } else if (s.health < 70 && (s.status === "injured" || s.status === "critical")) {
   593→      s.status = "injured";
   594→    } else if (s.health >= 70 && s.status !== "sick") {
   595→      s.status = "healthy";
   596→    }
   597→  });
   598→
   599→  // ---------- 6. Remove dead ----------
   600→  areaSurvivors.forEach((s) => {
   601→    if (s.health <= 0) {
   602→      survivorsDied.push(s.id);
   603→      newLog.push({ day, message: `[${area.name}] ${s.name} has died.`, type: "danger" });
   604→    }
   605→  });
   606→  area.survivorIds = area.survivorIds.filter((id) => {
   607→    const s = allSurvivors[id];
   608→    return s && s.health > 0;
   609→  });
   610→
   611→  // ---------- 7. Random bandit raid ----------
   612→  if (area.hasBase) {
   613→    const rng = makeRng(day * 9973 + 7 + area.hex[0] * 31 + area.hex[1] * 17);
   614→    if (day > 2 && rng() < 0.2) {
   615→      const raidPower = randInt(rng, 2, 4) + Math.floor(day / 3);
   616→      const defense =
   617→        area.buildings.watchtower.level * 15 +
   618→        areaSurvivors
   619→          .filter((s) => s.role === "idle")
   620→          .reduce((sum, s) => sum + s.skills.combat * 2, 0);
   621→
   622→      if (defense >= raidPower) {
   623→        resources.ammo = Math.max(0, resources.ammo - Math.min(resources.ammo, raidPower));
   624→        newLog.push({ day, message: `[${area.name}] Bandit raid repelled.`, type: "success" });
   625→      } else {
   626→        const stolenFood = Math.min(resources.food, randInt(rng, 5, 15));
   627→        const stolenWater = Math.min(resources.water, randInt(rng, 5, 15));
   628→        const stolenMats = Math.min(resources.materials, randInt(rng, 5, 15));
   629→        resources.food -= stolenFood;
   630→        resources.water -= stolenWater;
   631→        resources.materials -= stolenMats;
   632→        newLog.push({ day, message: `[${area.name}] Bandits broke through! Lost ${stolenFood} food, ${stolenWater} water, ${stolenMats} materials.`, type: "danger" });
   633→      }
   634→    }
   635→  }
   636→
   637→  // ---------- 8. Re-cap & save ----------
   638→  area.resourceCaps = calculateCaps(area);
   639→  area.resources = clampResources(resources, area.resourceCaps);
   640→
   641→  // Reset onMission survivors to idle
   642→  areaSurvivors.forEach((s) => {
   643→    if (s.role === "onMission") s.role = "idle";
   644→  });
   645→
   646→  return { survivorsDied };
   647→}

   649→// ---------- Process transfers for the day ----------
   650→function processTransfers(
   651→  state: GameState,
   652→  day: number,
   653→  newLog: GameLogEntry[]
   654→): Transfer[] {
   655→  const remaining: Transfer[] = [];
   656→  for (const transfer of state.transfers) {
   657→    if (transfer.arrivalDay <= day) {
   658→      // Arrived
   659→      const destArea = state.areas[transfer.toAreaId];
   660→      if (destArea) {
   661→        // If area is undiscovered, discover it now
   662→        if (!destArea.discovered) {
   663→          destArea.discovered = true;
   664→          destArea.locations = generateAreaLocations(
   665→            destArea.id,
   666→            destArea.type,
   667→            day * 1000 + destArea.hex[0] * 100 + destArea.hex[1]
   668→          );
   669→          newLog.push({
   670→            day,
   671→            message: `${destArea.name} has been discovered!`,
   672→            type: "success",
   673→          });
   674→        }
   675→        // Add survivors to destination area
   676→        for (const sid of transfer.survivorIds) {
   677→          if (!destArea.survivorIds.includes(sid)) {
   678→            destArea.survivorIds.push(sid);
   679→          }
   680→          // Reset their role
   681→          const s = state.survivors[sid];
   682→          if (s) {
   683→            s.role = "idle";
   684→            s.assignedTeamId = undefined;
   685→          }
   686→        }
   687→        // Add resources
   688→        for (const [k, v] of Object.entries(transfer.resources)) {
   689→          destArea.resources[k as ResourceType] += v as number;
   690→        }
   691→        destArea.resourceCaps = calculateCaps(destArea);
   692→        destArea.resources = clampResources(destArea.resources, destArea.resourceCaps);
   693→
   694→        const survNames = transfer.survivorIds
   695→          .map((id) => state.survivors[id]?.name)
   696→          .filter(Boolean)
   697→          .join(", ");
   698→        newLog.push({
   699→          day,
   700→          message: `Travelers arrived at ${destArea.name}${survNames ? `: ${survNames}` : ""}.`,
   701→          type: "info",
   702→        });
   703→      }
   704→    } else {
   705→      remaining.push(transfer);
   706→    }
   707→  }
   708→  return remaining;
   709→}
