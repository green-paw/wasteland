    74→// ---------- Low-poly terrain ----------
    75→function Terrain() {
    76→  const geometry = useMemo(() => {
    77→    const geo = new THREE.PlaneGeometry(80, 80, 40, 40);
    78→    // subtle height variation
    79→    const pos = geo.attributes.position;
    80→    for (let i = 0; i < pos.count; i++) {
    81→      const x = pos.getX(i);
    82→      const y = pos.getY(i);
    83→      const dist = Math.sqrt(x * x + y * y);
    84→      const noise = Math.sin(x * 0.3) * Math.cos(y * 0.3) * 0.3;
    85→      pos.setZ(i, noise + Math.sin(dist * 0.2) * 0.2);
    86→    }
    87→    geo.computeVertexNormals();
    88→    return geo;
    89→  }, []);
    90→
    91→  return (
    92→    <mesh
    93→      geometry={geometry}
    94→      rotation={[-Math.PI / 2, 0, 0]}
    95→      position={[0, -0.1, 0]}
    96→      receiveShadow
    97→    >
    98→      <meshStandardMaterial color="#3a4a2a" flatShading roughness={1} />
    99→    </mesh>
   100→  );
   101→}

   650→// ---------- Animated location marker ----------
   651→function LocationMarker({
   652→  location,
   653→  selected,
   654→  onClick,
   655→  hasMission,
   656→}: {
   657→  location: GameLocation;
   658→  selected: boolean;
   659→  onClick: () => void;
   660→  hasMission: boolean;
   661→}) {
   662→  const def = LOCATION_DEFS[location.type];
   663→  const ringRef = useRef<THREE.Mesh>(null);
   664→  const [hovered, setHovered] = useState(false);
   665→
   666→  useFrame((state) => {
   667→    if (ringRef.current) {
   668→      const t = state.clock.elapsedTime;
   669→      const scale = 1 + Math.sin(t * 2) * 0.15;
   670→      ringRef.current.scale.setScalar(scale);
   671→    }
   672→  });
   673→
   674→  const markerColor = location.salvageDepleted
   675→    ? "#6b7280" // grey for depleted
   676→    : location.cleared
   677→    ? "#22c55e" // green for cleared (still has salvage)
   678→    : hasMission
   679→    ? "#fbbf24"
   680→    : selected
   681→    ? "#ef4444"
   682→    : hovered
   683→    ? "#f87171"
   684→    : "#dc2626";
   685→
   686→  return (
   687→    <group position={location.position}>
   688→      {/* Invisible larger click-catcher for easier interaction (box for top-down view) */}
   689→      <mesh
   690→        position={[0, 1, 0]}
   691→        onClick={(e) => {
   692→          e.stopPropagation();
   693→          onClick();
   694→        }}
   695→        onPointerOver={(e) => {
   696→          e.stopPropagation();
   697→          setHovered(true);
   698→          document.body.style.cursor = "pointer";
   699→        }}
   700→        onPointerOut={() => {
   701→          setHovered(false);
   702→          document.body.style.cursor = "auto";
   703→        }}
   704→      >
   705→        <boxGeometry args={[3, 3, 3]} />
   706→        <meshBasicMaterial transparent opacity={0} depthWrite={false} />
   707→      </mesh>
   708→
   709→      {/* The building — faded if cleared */}
   710→      <group>
   711→        <LocationBuilding type={location.type} />
   712→        {location.cleared && (
   713→          // Overlay a translucent green tint on cleared buildings
   714→          <mesh position={[0, 0, 0]}>
   715→            <boxGeometry args={[2.5, 2.5, 2.5]} />
   716→            <meshBasicMaterial
   717→              color="#22c55e"
   718→              transparent
   719→              opacity={0.15}
   720→              depthWrite={false}
   721→            />
   722→          </mesh>
   723→        )}
   724→      </group>
   725→
   726→      {/* Selection ring (animated) — for selected/hovered/mission */}
   727→      {(selected || hovered || hasMission) && (
   728→        <mesh
   729→          ref={ringRef}
   730→          rotation={[-Math.PI / 2, 0, 0]}
   731→          position={[0, 0.05, 0]}
   732→        >
   733→          <ringGeometry args={[1.5, 1.7, 24]} />
   734→          <meshBasicMaterial
   735→            color={markerColor}
   736→            transparent
   737→            opacity={0.8}
   738→            side={THREE.DoubleSide}
   739→          />
   740→        </mesh>
   741→      )}
   742→
   743→      {/* Always-on small dot */}
   744→      <mesh position={[0, 0.1, 0]} rotation={[-Math.PI / 2, 0, 0]}>
   745→        <circleGeometry args={[0.3, 16]} />
   746→        <meshBasicMaterial color={markerColor} />
   747→      </mesh>
   748→
   749→      {/* Cleared indicator — green ring on the ground + floating checkmark flag */}
   750→      {location.cleared && (
   751→        <>
   752→          {/* Persistent green ring on the ground (always visible from top) */}
   753→          <mesh
   754→            rotation={[-Math.PI / 2, 0, 0]}
   755→            position={[0, 0.08, 0]}
   756→          >
   757→            <ringGeometry args={[2.0, 2.3, 24]} />
   758→            <meshBasicMaterial
   759→              color="#22c55e"
   760→              transparent
   761→              opacity={0.6}
   762→              side={THREE.DoubleSide}
   763→            />
   764→          </mesh>
   765→          {/* Floating flag with checkmark (visible from any angle) */}
   766→          <group position={[0, 3.5, 0]}>
   767→            {/* Pole */}
   768→            <mesh position={[0, -0.6, 0]}>
   769→              <cylinderGeometry args={[0.04, 0.04, 1.2, 6]} />
   770→              <meshBasicMaterial color="#1a1a1a" />
   771→            </mesh>
   772→            {/* Flag */}
   773→            <mesh position={[0.3, -0.1, 0]}>
   774→              <planeGeometry args={[0.6, 0.4]} />
   775→              <meshBasicMaterial color="#22c55e" side={THREE.DoubleSide} />
   776→            </mesh>
   777→          </group>
   778→        </>
   779→      )}
   780→
   781→      {/* Hover label — fixed screen size, follows the 3D position.
   782→          Only shown while actively hovering (not when selected), so the
   783→          tooltip disappears after the player clicks the building. */}
   784→      {hovered && (
   785→        <Html
   786→          position={[0, 3.5, 0]}
   787→          center
   788→          prepend
   789→          zIndexRange={[100, 0]}
   790→          style={{ pointerEvents: "none" }}
   791→        >
   792→          <div className="bg-stone-950/95 border border-stone-700 rounded px-3 py-2 text-xs text-stone-100 whitespace-nowrap shadow-xl">
   793→            <div className="font-bold text-amber-100 text-sm">{def.label}</div>
   794→            <div className="text-[11px] text-stone-400 mt-0.5">
   795→              {location.cleared ? (
   796→                <span className="text-emerald-400">✓ Cleared</span>
   797→              ) : (
   798→                <>
   799→                  {ENEMY_INFO[location.enemyType].icon} {location.enemyCount}{" "}
   800→                  {ENEMY_INFO[location.enemyType].label}
   801→                </>
   802→              )}
   803→            </div>
   804→            {!location.cleared &&
   805→              Object.keys(location.loot).length > 0 && (
   806→                <div className="mt-1.5 pt-1.5 border-t border-stone-800">
   807→                  <div className="text-[9px] uppercase tracking-wide text-stone-500 mb-0.5">
   808→                    Expected Loot
   809→                  </div>
   810→                  <div className="flex items-center gap-2 flex-wrap">
   811→                    {Object.entries(location.loot).map(([k, v]) => (
   812→                      <span
   813→                        key={k}
   814→                        className="flex items-center gap-0.5 text-[11px] text-stone-200"
   815→                      >
   816→                        <span>
   817→                          {
   818→                            RESOURCE_INFO[k as keyof typeof RESOURCE_INFO]
   819→                              .icon
   820→                          }
   821→                        </span>
   822→                        <span className="font-medium">{v}</span>
   823→                      </span>
   824→                    ))}
   825→                  </div>
   826→                </div>
   827→              )}
   828→            {location.cleared &&
   829→              !location.salvageDepleted &&
   830→              Object.keys(location.salvagePool).length > 0 && (
   831→                <div className="mt-1.5 pt-1.5 border-t border-stone-800">

   837→                      <span
   838→                        key={k}
   839→                        className="flex items-center gap-0.5 text-[11px] text-emerald-200"
   840→                      >
   841→                        <span>
   842→                          {
   843→                            RESOURCE_INFO[k as keyof typeof RESOURCE_INFO]
   844→                              .icon
   845→                          }
   846→                        </span>
   847→                        <span className="font-medium">{v as number}</span>

   858→              </div>

   860→          </div>

   863→    </group>

   867→// ---------- Mission Lines ----------
   868→function MissionLines() {
   869→  const missions = useGameStore((s) => s.missions);
   870→  const locations = useGameStore((s) => s.locations);
   871→  const teams = useGameStore((s) => s.teams);
   872→
   873→  const pendingMissions = missions.filter((m) => m.status === "pending");
   874→
   875→  return (
   876→    <>
   877→      {pendingMissions.map((m) => {
   878→        const loc = locations.find((l) => l.id === m.locationId);
   879→        if (!loc) return null;
   880→        const team = teams.find((t) => t.id === m.teamId);
   881→        // Salvage missions use a green dashed line, scout missions use team color
   882→        const isSalvage = m.missionType === "salvage";
   883→        const teamColor = isSalvage
   884→          ? "#22c55e"
   885→          : team?.name?.toLowerCase().includes("alpha")
   886→          ? "#fbbf24"
   887→          : team?.name?.toLowerCase().includes("bravo")
   888→          ? "#60a5fa"
   889→          : "#a78bfa";
   890→        return (
   891→          <Line
   892→            key={m.id}
   893→            points={[
   894→              [0, 0.5, 0],
   895→              [loc.position[0], 0.5, loc.position[2]],
   896→            ]}
   897→            color={teamColor}
   898→            lineWidth={2}
   899→            dashed
   900→            dashSize={0.5}
   901→            gapSize={0.3}
   902→          />
   903→        );
   904→      })}
   905→    </>
   906→  );
   907→}
