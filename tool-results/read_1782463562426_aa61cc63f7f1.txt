    74→    74→// ---------- Low-poly terrain ----------
    75→    75→function Terrain() {
    76→    76→  const geometry = useMemo(() => {
    77→    77→    const geo = new THREE.PlaneGeometry(80, 80, 40, 40);
    78→    78→    // subtle height variation
    79→    79→    const pos = geo.attributes.position;
    80→    80→    for (let i = 0; i < pos.count; i++) {
    81→    81→      const x = pos.getX(i);
    82→    82→      const y = pos.getY(i);
    83→    83→      const dist = Math.sqrt(x * x + y * y);
    84→    84→      const noise = Math.sin(x * 0.3) * Math.cos(y * 0.3) * 0.3;
    85→    85→      pos.setZ(i, noise + Math.sin(dist * 0.2) * 0.2);
    86→    86→    }
    87→    87→    geo.computeVertexNormals();
    88→    88→    return geo;
    89→    89→  }, []);
    90→    90→
    91→    91→  return (
    92→    92→    <mesh
    93→    93→      geometry={geometry}
    94→    94→      rotation={[-Math.PI / 2, 0, 0]}
    95→    95→      position={[0, -0.1, 0]}
    96→    96→      receiveShadow
    97→    97→    >
    98→    98→      <meshStandardMaterial color="#3a4a2a" flatShading roughness={1} />
    99→    99→    </mesh>
   100→   100→  );
   101→   101→}

   651→   651→function LocationMarker({
   652→   652→  location,
