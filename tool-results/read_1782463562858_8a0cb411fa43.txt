    45→    45→function clampResources(res: Resources, caps: Resources): Resources {
    46→    46→  const out: Resources = { ...res };
    47→    47→  (Object.keys(out) as ResourceType[]).forEach((k) => {
    48→    48→    out[k] = Math.max(0, Math.min(caps[k], out[k]));
    49→    49→  });
    50→    50→  return out;

    53→    53→function canAfford(res: Resources, cost: Partial<Resources>): boolean {
    54→    54→  for (const [k, v] of Object.entries(cost)) {
    55→    55→    if (res[k as ResourceType] < (v as number)) return false;
    56→    56→  }
    57→    57→  return true;
    58→    58→}

    60→    60→function subtractCost(res: Resources, cost: Partial<Resources>): Resources {
    61→    61→  const out = { ...res };
    62→    62→  for (const [k, v] of Object.entries(cost)) {
    63→    63→    out[k as ResourceType] -= v as number;
    64→    64→  }
    65→    65→  return out;
    66→    66→}

   179→   179→// ---------- Resolve a scout Mission (combat + loot) ----------
   180→   180→function resolveMission(
   181→   181→  mission: Mission,
   182→   182→  survivors: Survivor[],
   183→   183→  location: GameLocation,
   184→   184→  day: number
   185→   185→): { result: MissionResult; log: GameLogEntry[] } {
   186→   186→  const log: GameLogEntry[] = [];
   187→   187→  const teamSurvivors = survivors.filter((s) => mission.team.includes(s.id));
   188→   188→  const rng = makeRng(
   189→   189→    mission.id
   190→   190→      .split("")
   191→   191→      .reduce((a, c) => a + c.charCodeAt(0), 0) +
   192→   192→      day * 7919 +
   193→   193→      location.position[0] * 31
   194→   194→  );
   195→   195→
   196→   196→  const result: MissionResult = {
   197→   197→    success: true,
   198→   198→    lootGained: {},
   199→   199→    survivorsRecruited: [],
   200→   200→    casualties: [],
   201→   201→    injuries: [],
   202→   202→    log: [],
   203→   203→  };
   204→   204→
   205→   205→  if (!location.cleared && location.enemyCount > 0) {
   206→   206→    const enemyDef = ENEMY_INFO[location.enemyType];
   207→   207→    const teamCombat =
   208→   208→      teamSurvivors.reduce((sum, s) => sum + s.skills.combat, 0) +
   209→   209→      teamSurvivors.reduce((sum, s) => sum + (s.health > 50 ? 1 : 0), 0) * 0.5;
   210→   210→    const enemyPower = location.enemyCount * enemyDef.combatPower;
   211→   211→
   212→   212→    result.log.push(
   213→   213→      `Team encountered ${location.enemyCount} ${enemyDef.label} at ${location.name}.`
   214→   214→    );
   215→   215→
   216→   216→    if (teamCombat >= enemyPower) {
   217→   217→      result.success = true;
   218→   218→      result.log.push(
   219→   219→        `Victory! Team combat ${teamCombat.toFixed(1)} vs enemy ${enemyPower.toFixed(1)}.`
   220→   220→      );
   221→   221→      teamSurvivors.forEach((s) => {
   222→   222→        const injuryChance = 0.15 + location.dangerLevel * 0.05;
   223→   223→        if (rng() < injuryChance) {
   224→   224→          const dmg = randInt(rng, 10, 30);
   225→   225→          s.health = Math.max(1, s.health - dmg);
   226→   226→          s.status = s.health < 30 ? "critical" : "injured";
   227→   227→          result.injuries.push(s.id);
   228→   228→          result.log.push(`${s.name} was injured (-${dmg} HP).`);
   229→   229→        }
   230→   230→      });
   231→   231→    } else {
   232→   232→      result.success = false;
   233→   233→      result.log.push(
   234→   234→        `Defeat! Team combat ${teamCombat.toFixed(1)} vs enemy ${enemyPower.toFixed(1)}.`
   235→   235→      );
   236→   236→      teamSurvivors.forEach((s) => {
   237→   237→        const deathChance = 0.1 + Math.max(0, enemyPower - teamCombat) * 0.05;
   238→   238→        if (rng() < deathChance) {
   239→   239→          result.casualties.push(s.id);
   240→   240→          result.log.push(`${s.name} was killed in action.`);
   241→   241→        } else {
   242→   242→          const dmg = randInt(rng, 25, 55);
   243→   243→          s.health = Math.max(5, s.health - dmg);
   244→   244→          s.status = s.health < 30 ? "critical" : "injured";
   245→   245→          result.injuries.push(s.id);
   246→   246→          result.log.push(`${s.name} was badly injured (-${dmg} HP).`);
   247→   247→        }
   248→   248→      });
   249→   249→      const halfLoot: Partial<Resources> = {};
   250→   250→      for (const [k, v] of Object.entries(location.loot)) {
   251→   251→        halfLoot[k as ResourceType] = Math.floor((v as number) * 0.4);
   252→   252→      }
   253→   253→      result.lootGained = halfLoot;
   254→   254→    }
   255→   255→  } else {
   256→   256→    result.log.push(`No hostiles at ${location.name}.`);
   257→   257→  }
   258→   258→
   259→   259→  if (result.success && !location.cleared) {
   260→   260→    const scavenging = teamSurvivors.reduce(
   261→   261→      (sum, s) => sum + s.skills.scavenging,
   262→   262→      0
   263→   263→    );
   264→   264→    const lootMultiplier = 0.5 + scavenging * 0.15 + rng() * 0.3;
   265→   265→    const loot: Partial<Resources> = {};
   266→   266→    for (const [k, v] of Object.entries(location.loot)) {
   267→   267→      loot[k as ResourceType] = Math.max(
   268→   268→        0,
   269→   269→        Math.round((v as number) * lootMultiplier)
   270→   270→      );
   271→   271→    }
   272→   272→    result.lootGained = { ...result.lootGained, ...loot };
   273→   273→    const lootSummary = Object.entries(loot)
   274→   274→      .filter(([, v]) => (v as number) > 0)
   275→   275→      .map(([k, v]) => `${v} ${k}`)
   276→   276→      .join(", ");
   277→   277→    if (lootSummary) {
   278→   278→      result.log.push(`Salvaged: ${lootSummary}.`);
   279→   279→    }
   280→   280→  }
   281→   281→
   282→   282→  if (result.success && !location.cleared) {
   283→   283→    if (rng() < location.survivorChance) {
   284→   284→      const newSurvivor = generateSurvivor(
   285→   285→        Math.floor(rng() * 1000000) + day * 13
   286→   286→      );
   287→   287→      newSurvivor.health = randInt(rng, 50, 90);
   288→   288→      newSurvivor.hunger = randInt(rng, 30, 70);
   289→   289→      newSurvivor.thirst = randInt(rng, 30, 70);
   290→   290→      newSurvivor.morale = randInt(rng, 30, 70);
   291→   291→      result.survivorsRecruited.push(newSurvivor);
   292→   292→      result.log.push(
   293→   293→        `Found a survivor named ${newSurvivor.name}! They joined your group.`
   294→   294→      );
   295→   295→    }
   296→   296→  }
   297→   297→
   298→   298→  location.explored = true;
   299→   299→  if (result.success) {
   300→   300→    location.cleared = true;
   301→   301→    location.enemyCount = 0;
   302→   302→  }
   303→   303→
   304→   304→  result.log.forEach((msg) =>
   305→   305→    log.push({
   306→   306→      day,
   307→   307→      message: msg,
   308→   308→      type: msg.includes("killed")
   309→   309→        ? "danger"
   310→   310→        : msg.includes("injured") || msg.includes("Defeat")
   311→   311→        ? "warning"
   312→   312→        : msg.includes("Victory") ||
   313→   313→          msg.includes("Salvaged") ||
   314→   314→          msg.includes("Found")
   315→   315→        ? "success"
   316→   316→        : "info",
   317→   317→    })
   318→   318→  );
   319→   319→
   320→   320→  return { result, log };
   321→   321→}

   323→   323→// ---------- Process a single area for the day ----------
   324→   324→function processArea(
   325→   325→  area: Area,
   326→   326→  allSurvivors: Record<string, Survivor>,
   327→   327→  day: number,
   328→   328→  newLog: GameLogEntry[]
   329→   329→): { survivorsDied: string[] } {
   330→   330→  const survivorsDied: string[] = [];
   331→   331→
   332→   332→  if (!area.hasBase && area.survivorIds.length === 0 && area.missions.length === 0) {
   333→   333→    return { survivorsDied };
   334→   334→  }
   335→   335→
   336→   336→  const areaSurvivors = area.survivorIds
   337→   337→    .map((id) => allSurvivors[id])
   338→   338→    .filter((s) => s !== undefined);
   339→   339→  let resources = { ...area.resources };
   340→   340→
   341→   341→  newLog.push({
   342→   342→    day,
   343→   343→    message: `--- ${area.name} (Day ${day}) ---`,
   344→   344→    type: "info",
   345→   345→  });
   346→   346→
   347→   347→  // ---------- 1. Resolve pending missions ----------
   348→   348→  area.missions = area.missions.map((m) => {
   349→   349→    if (m.status !== "pending") return m;
   350→   350→    const location = area.locations.find((l) => l.id === m.locationId);
   351→   351→    if (!location) return { ...m, status: "completed" as const };
   352→   352→
   353→   353→    const teamSurvivors = m.team
   354→   354→      .map((id) => allSurvivors[id])
   355→   355→      .filter((s) => s !== undefined);
   356→   356→
   357→   357→    if (m.missionType !== "salvage") {
   358→   358→      teamSurvivors.forEach((s) => {
   359→   359→        s.role = "onMission";
   360→   360→      });
   361→   361→    }
   362→   362→
   363→   363→    // -------- Salvage mission --------
   364→   364→    if (m.missionType === "salvage") {
   365→   365→      const salvageRng = makeRng(
   366→   366→        m.id.split("").reduce((a, c) => a + c.charCodeAt(0), 0) + day * 7919
   367→   367→      );
   368→   368→      const totalScavenge = teamSurvivors.reduce(
   369→   369→        (sum, s) => sum + s.skills.scavenging,
   370→   370→        0
   371→   371→      );
   372→   372→      const salvageWork = totalScavenge + teamSurvivors.length * 2;
   373→   373→      const poolEntries = Object.entries(location.salvagePool).filter(
   374→   374→        ([, v]) => (v as number) > 0
   375→   375→      ) as [ResourceType, number][];
   376→   376→
   377→   377→      const lootGained: Partial<Resources> = {};
   378→   378→      let depletedThisRun = false;
   379→   379→
   380→   380→      if (poolEntries.length === 0) {
   381→   381→        location.salvageDepleted = true;
   382→   382→        depletedThisRun = true;
   383→   383→      } else {
   384→   384→        const hasMaterials = poolEntries.some(([k]) => k === "materials");
   385→   385→        const otherEntries = poolEntries.filter(([k]) => k !== "materials");
   386→   386→        const allocations: [ResourceType, number][] = [];
   387→   387→        if (hasMaterials) {
   388→   388→          allocations.push(["materials", salvageWork * 0.5]);
   389→   389→        }
   390→   390→        if (otherEntries.length > 0) {
   391→   391→          const perType = (salvageWork * (hasMaterials ? 0.5 : 1)) / otherEntries.length;
   392→   392→          for (const [k] of otherEntries) {
   393→   393→            allocations.push([k, perType]);
   394→   394→          }
   395→   395→        }
   396→   396→        for (const [k, work] of allocations) {
   397→   397→          const remaining = location.salvagePool[k] ?? 0;
   398→   398→          if (remaining <= 0) continue;
   399→   399→          const yield_ = Math.max(1, Math.round(work * (0.8 + salvageRng() * 0.4)));
   400→   400→          const extracted = Math.min(remaining, yield_);
   401→   401→          location.salvagePool[k] = remaining - extracted;
   402→   402→          lootGained[k] = extracted;
   403→   403→        }
   404→   404→        const totalRemaining = Object.values(location.salvagePool).reduce(
   405→   405→          (sum, v) => sum + (v as number),
   406→   406→          0
   407→   407→        );
   408→   408→        if (totalRemaining <= 0) {
   409→   409→          location.salvageDepleted = true;
   410→   410→          depletedThisRun = true;
   411→   411→        }
   412→   412→        const lootSummary = Object.entries(lootGained)
   413→   413→          .filter(([, v]) => (v as number) > 0)
   414→   414→          .map(([k, v]) => `${v} ${k}`)
   415→   415→          .join(", ");
   416→   416→        if (lootSummary) {
   417→   417→          newLog.push({
   418→   418→            day,
   419→   419→            message: `[${area.name}] Salvaged: ${lootSummary}.`,
   420→   420→            type: "success",
   421→   421→          });
   422→   422→        }
   423→   423→        if (depletedThisRun) {
   424→   424→          newLog.push({
   425→   425→            day,
   426→   426→            message: `[${area.name}] ${location.name} fully salvaged.`,
   427→   427→            type: "warning",
   428→   428→          });
   429→   429→        }
   430→   430→      }
   431→   431→
   432→   432→      for (const [k, v] of Object.entries(lootGained)) {
   433→   433→        resources[k as ResourceType] += v as number;
   434→   434→      }
   435→   435→
   436→   436→      if (depletedThisRun) {
   437→   437→        teamSurvivors.forEach((s) => {
   438→   438→          s.role = "idle";
   439→   439→          s.assignedTeamId = undefined;
   440→   440→        });
   441→   441→        const team = area.teams.find((t) => t.id === m.teamId);
   442→   442→        if (team) team.locationId = null;
   443→   443→        return { ...m, status: "completed" as const };
   444→   444→      }
   445→   445→      return m; // stays pending
   446→   446→    }
   447→   447→
   448→   448→    // -------- Scout mission --------
   449→   449→    const { result, log } = resolveMission(m, teamSurvivors, location, day);
   450→   450→    newLog.push(...log);
   451→   451→
   452→   452→    for (const [k, v] of Object.entries(result.lootGained)) {
   453→   453→      resources[k as ResourceType] += v as number;
   454→   454→    }
   455→   455→
   456→   456→    result.casualties.forEach((id) => survivorsDied.push(id));
   457→   457→    result.survivorsRecruited.forEach((ns) => {
   458→   458→      allSurvivors[ns.id] = ns;
   459→   459→      area.survivorIds.push(ns.id);
   460→   460→    });
   461→   461→
   462→   462→    teamSurvivors.forEach((s) => {
   463→   463→      const updated = allSurvivors[s.id];
   464→   464→      if (updated) {
   465→   465→        updated.role = "idle";
   466→   466→        updated.assignedTeamId = undefined;
   467→   467→      }
   468→   468→    });
   469→   469→
   470→   470→    const team = area.teams.find((t) => t.id === m.teamId);
   471→   471→    if (team) team.locationId = null;
   472→   472→
   473→   473→    return { ...m, status: "completed" as const, result };
   474→   474→  });
   475→   475→
   476→   476→  area.missions = area.missions.filter((m) => m.status === "pending");
   477→   477→
   478→   478→  // ---------- 2. Building production ----------
   479→   479→  if (area.hasBase) {
   480→   480→    const farmFood = area.buildings.farm.level * 5;
   481→   481→    const wellWater = area.buildings.well.level * 5;
   482→   482→    if (farmFood > 0) {
   483→   483→      resources.food += farmFood;
   484→   484→      newLog.push({ day, message: `[${area.name}] Farm produced ${farmFood} food.`, type: "success" });
   485→   485→    }
   486→   486→    if (wellWater > 0) {
   487→   487→      resources.water += wellWater;
   488→   488→      newLog.push({ day, message: `[${area.name}] Well produced ${wellWater} water.`, type: "success" });
   489→   489→    }
   490→   490→  }
   491→   491→
   492→   492→  // ---------- 3. Survivor consumption ----------
   493→   493→  const consumption = areaSurvivors.length;
   494→   494→  const foodNeeded = consumption;
   495→   495→  const waterNeeded = consumption;
   496→   496→  let foodShortage = 0;
   497→   497→  let waterShortage = 0;
   498→   498→
   499→   499→  if (resources.food >= foodNeeded) {
   500→   500→    resources.food -= foodNeeded;
   501→   501→  } else {
   502→   502→    foodShortage = foodNeeded - resources.food;
   503→   503→    resources.food = 0;
   504→   504→  }
   505→   505→  if (resources.water >= waterNeeded) {
   506→   506→    resources.water -= waterNeeded;
   507→   507→  } else {
   508→   508→    waterShortage = waterNeeded - resources.water;
   509→   509→    resources.water = 0;
   510→   510→  }
   511→   511→
   512→   512→  if (foodShortage > 0) {
   513→   513→    newLog.push({ day, message: `[${area.name}] Food shortage! ${foodShortage} survivors go hungry.`, type: "warning" });
   514→   514→  } else if (consumption > 0) {
   515→   515→    newLog.push({ day, message: `[${area.name}] Survivors consumed ${foodNeeded} food and ${waterNeeded} water.`, type: "info" });
   516→   516→  }
   517→   517→  if (waterShortage > 0) {
   518→   518→    newLog.push({ day, message: `[${area.name}] Water shortage! ${waterShortage} survivors are dehydrated.`, type: "warning" });
   519→   519→  }
   520→   520→
   521→   521→  // ---------- 4. Update survivor stats ----------
   522→   522→  areaSurvivors.forEach((s) => {
   523→   523→    if (foodShortage === 0 && s.role !== "onMission") {
   524→   524→      s.hunger = Math.max(0, s.hunger - 30);
   525→   525→    } else if (s.role === "onMission") {
   526→   526→      s.hunger = Math.min(100, s.hunger + 10);
   527→   527→    } else {
   528→   528→      s.hunger = Math.min(100, s.hunger + 25);
   529→   529→      s.health = Math.max(0, s.health - 5);
   530→   530→    }
   531→   531→
   532→   532→    if (waterShortage === 0 && s.role !== "onMission") {
   533→   533→      s.thirst = Math.max(0, s.thirst - 35);
   534→   534→    } else if (s.role === "onMission") {
   535→   535→      s.thirst = Math.min(100, s.thirst + 15);
   536→   536→    } else {
   537→   537→      s.thirst = Math.min(100, s.thirst + 30);
   538→   538→      s.health = Math.max(0, s.health - 8);
   539→   539→    }
   540→   540→
   541→   541→    if (s.hunger >= 90) {
   542→   542→      s.health = Math.max(0, s.health - 8);
   543→   543→      newLog.push({ day, message: `[${area.name}] ${s.name} is starving.`, type: "warning" });
   544→   544→    }
   545→   545→    if (s.thirst >= 90) {
   546→   546→      s.health = Math.max(0, s.health - 10);
   547→   547→      newLog.push({ day, message: `[${area.name}] ${s.name} is severely dehydrated.`, type: "warning" });
   548→   548→    }
   549→   549→
   550→   550→    if (s.role === "resting") {
   551→   551→      s.morale = Math.min(100, s.morale + 10);
   552→   552→      s.health = Math.min(100, s.health + 5);
   553→   553→    }
   554→   554→    if (s.health < 30) {
   555→   555→      s.morale = Math.max(0, s.morale - 5);
   556→   556→    }
   557→   557→  });
   558→   558→
   559→   559→  // ---------- 5. Healing ----------
   560→   560→  if (area.hasBase && area.buildings.infirmary.level > 0) {
   561→   561→    const heal = area.buildings.infirmary.level * 10;
   562→   562→    areaSurvivors.forEach((s) => {
   563→   563→      if (s.status !== "healthy" && s.health < 100) {
   564→   564→        if (resources.medicine >= 1) {
   565→   565→          resources.medicine -= 1;
   566→   566→          s.health = Math.min(100, s.health + heal);
   567→   567→          if (s.health >= 70) s.status = "healthy";
   568→   568→          else if (s.health >= 40 && s.status === "critical") s.status = "injured";
   569→   569→          newLog.push({ day, message: `[${area.name}] ${s.name} treated at infirmary (+${heal} HP).`, type: "success" });
   570→   570→        }
   571→   571→      }
   572→   572→    });
   573→   573→  } else {
   574→   574→    areaSurvivors.forEach((s) => {
   575→   575→      if (s.status !== "healthy" && s.health < 100) {
   576→   576→        const naturalHeal = 3;
   577→   577→        s.health = Math.min(100, s.health + naturalHeal);
   578→   578→        if (s.health >= 70) s.status = "healthy";
   579→   579→        else if (s.health >= 40 && s.status === "critical") s.status = "injured";
   580→   580→      }
   581→   581→    });
   582→   582→  }
   583→   583→
   584→   584→  // Update status
   585→   585→  areaSurvivors.forEach((s) => {
   586→   586→    if (s.health <= 0) {
   587→   587→      // dead
   588→   588→    } else if (s.health < 25) {
   589→   589→      s.status = "critical";
   590→   590→    } else if (s.health < 50) {
   591→   591→      s.status = s.status === "sick" ? "sick" : "injured";
   592→   592→    } else if (s.health < 70 && (s.status === "injured" || s.status === "critical")) {
   593→   593→      s.status = "injured";
   594→   594→    } else if (s.health >= 70 && s.status !== "sick") {
   595→   595→      s.status = "healthy";
   596→   596→    }
   597→   597→  });
   598→   598→
   599→   599→  // ---------- 6. Remove dead ----------
   600→   600→  areaSurvivors.forEach((s) => {
   601→   601→    if (s.health <= 0) {
   602→   602→      survivorsDied.push(s.id);
   603→   603→      newLog.push({ day, message: `[${area.name}] ${s.name} has died.`, type: "danger" });
   604→   604→    }
   605→   605→  });
   606→   606→  area.survivorIds = area.survivorIds.filter((id) => {
   607→   607→    const s = allSurvivors[id];
   608→   608→    return s && s.health > 0;
   609→   609→  });
   610→   610→
   611→   611→  // ---------- 7. Random bandit raid ----------
   612→   612→  if (area.hasBase) {
   613→   613→    const rng = makeRng(day * 9973 + 7 + area.hex[0] * 31 + area.hex[1] * 17);
   614→   614→    if (day > 2 && rng() < 0.2) {
   615→   615→      const raidPower = randInt(rng, 2, 4) + Math.floor(day / 3);
   616→   616→      const defense =
   617→   617→        area.buildings.watchtower.level * 15 +
   618→   618→        areaSurvivors
   619→   619→          .filter((s) => s.role === "idle")
   620→   620→          .reduce((sum, s) => sum + s.skills.combat * 2, 0);
   621→   621→
   622→   622→      if (defense >= raidPower) {
   623→   623→        resources.ammo = Math.max(0, resources.ammo - Math.min(resources.ammo, raidPower));
   624→   624→        newLog.push({ day, message: `[${area.name}] Bandit raid repelled.`, type: "success" });
   625→   625→      } else {
   626→   626→        const stolenFood = Math.min(resources.food, randInt(rng, 5, 15));
   627→   627→        const stolenWater = Math.min(resources.water, randInt(rng, 5, 15));
   628→   628→        const stolenMats = Math.min(resources.materials, randInt(rng, 5, 15));
   629→   629→        resources.food -= stolenFood;
   630→   630→        resources.water -= stolenWater;
   631→   631→        resources.materials -= stolenMats;
   632→   632→        newLog.push({ day, message: `[${area.name}] Bandits broke through! Lost ${stolenFood} food, ${stolenWater} water, ${stolenMats} materials.`, type: "danger" });
   633→   633→      }
   634→   634→    }
   635→   635→  }
   636→   636→
   637→   637→  // ---------- 8. Re-cap & save ----------
   638→   638→  area.resourceCaps = calculateCaps(area);
   639→   639→  area.resources = clampResources(resources, area.resourceCaps);
   640→   640→
   641→   641→  // Reset onMission survivors to idle
   642→   642→  areaSurvivors.forEach((s) => {
   643→   643→    if (s.role === "onMission") s.role = "idle";
   644→   644→  });
   645→   645→
   646→   646→  return { survivorsDied };
   647→   647→}

   649→   649→// ---------- Process transfers for the day ----------
   650→   650→function processTransfers(
   651→   651→  state: GameState,
   652→   652→  day: number,
   653→   653→  newLog: GameLogEntry[]
   654→   654→): Transfer[] {
   655→   655→  const remaining: Transfer[] = [];
   656→   656→  for (const transfer of state.transfers) {
   657→   657→    if (transfer.arrivalDay <= day) {
   658→   658→      // Arrived
   659→   659→      const destArea = state.areas[transfer.toAreaId];
   660→   660→      if (destArea) {
   661→   661→        // If area is undiscovered, discover it now
   662→   662→        if (!destArea.discovered) {
   663→   663→          destArea.discovered = true;
   664→   664→          destArea.locations = generateAreaLocations(
   665→   665→            destArea.id,
   666→   666→            destArea.type,
   667→   667→            day * 1000 + destArea.hex[0] * 100 + destArea.hex[1]
   668→   668→          );
   669→   669→          newLog.push({
   670→   670→            day,
   671→   671→            message: `${destArea.name} has been discovered!`,
   672→   672→            type: "success",
   673→   673→          });
   674→   674→        }
   675→   675→        // Add survivors to destination area
   676→   676→        for (const sid of transfer.survivorIds) {
   677→   677→          if (!destArea.survivorIds.includes(sid)) {
   678→   678→            destArea.survivorIds.push(sid);
   679→   679→          }
   680→   680→          // Reset their role
   681→   681→          const s = state.survivors[sid];
   682→   682→          if (s) {
   683→   683→            s.role = "idle";
   684→   684→            s.assignedTeamId = undefined;
   685→   685→          }
   686→   686→        }
   687→   687→        // Add resources
   688→   688→        for (const [k, v] of Object.entries(transfer.resources)) {
   689→   689→          destArea.resources[k as ResourceType] += v as number;
   690→   690→        }
   691→   691→        destArea.resourceCaps = calculateCaps(destArea);
   692→   692→        destArea.resources = clampResources(destArea.resources, destArea.resourceCaps);
   693→   693→
   694→   694→        const survNames = transfer.survivorIds
   695→   695→          .map((id) => state.survivors[id]?.name)
   696→   696→          .filter(Boolean)
   697→   697→          .join(", ");
   698→   698→        newLog.push({
   699→   699→          day,
   700→   700→          message: `Travelers arrived at ${destArea.name}${survNames ? `: ${survNames}` : ""}.`,
   701→   701→          type: "info",
   702→   702→        });
   703→   703→      }
   704→   704→    } else {
   705→   705→      remaining.push(transfer);
   706→   706→    }
   707→   707→  }
   708→   708→  return remaining;
   709→   709→}
